def solve_tribanded(ab, b):
    a = ab.copy()
    x = b.copy()
    n = b.size
    # eliminação
    for i in range(1,n):
        w = a[2,i-1]/a[1,i-1]
        a[1,i] -= w * a[0,i]
        x[i] -= w * x[i-1]
    # resolve
    x[n-1] = x[n-1]/a[1,n-1]
    for i in range(n-2,-1,-1):
        x[i] = (x[i] - a[0,i+1]*x[i+1])/a[1,i]
    return x
